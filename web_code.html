Practical 06 - Dialog Boxes

Demonstrate the JavaScript dialog boxes: alert, prompt and confirm, and show the result on the page.

Save as dialogs.html 
<!DOCTYPE html>
<html>
<head>
<title>Dialog Boxes</title>
<script>
function showAlert() {
  alert("This is an Alert Box!");
}

function showPrompt() {
  var name = prompt("Enter your name:");
  if (name != null)
    document.getElementById("result").innerHTML = "You entered: " + name;
}

function showConfirm() {
  var ans = confirm("Do you want to continue?");
  if (ans)
    document.getElementById("result").innerHTML = "You clicked OK";
  else
    document.getElementById("result").innerHTML = "You clicked Cancel";
}
</script>
</head>
<body>

<h2>Dialog Boxes</h2>

<button onclick="showAlert()">Alert Box</button>
<button onclick="showPrompt()">Prompt Box</button>
<button onclick="showConfirm()">Confirm Box</button>

<p id="result"></p>

</body>
</html>

Practical 07 - Registration Form Validation

Create a student registration form and validate the name, email and 10-digit mobile number using JavaScript regular expressions before submitting.

Save as registration.html 
<!DOCTYPE html>
<html>
<head>
<title>Student Registration Form</title>
<script>
function validateForm() {
  var name = document.getElementById("name").value;
  var email = document.getElementById("email").value;
  var mobile = document.getElementById("mobile").value;

  var namePattern = /^[A-Za-z ]{3,}$/;
  var emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  var mobilePattern = /^[0-9]{10}$/;

  if (!namePattern.test(name)) {
    alert("Name must contain only letters and minimum 3 characters.");
    return false;
  }
  if (!emailPattern.test(email)) {
    alert("Please enter a valid email address.");
    return false;
  }
  if (!mobilePattern.test(mobile)) {
    alert("Mobile number must be exactly 10 digits.");
    return false;
  }
  alert("Registration Successful!");
  return true;
}
</script>
</head>
<body>

<h2>Student Registration Form</h2>

<form onsubmit="return validateForm()">

Name: <input type="text" id="name" name="name"><br><br>
Email: <input type="text" id="email" name="email"><br><br>
Mobile: <input type="text" id="mobile" name="mobile"><br><br>
<input type="submit" value="Register">

</form>

</body>
</html>

Practical 10 - PHP Variables and Loops

Write a PHP script to demonstrate variables, arithmetic operators, if-else, while, for and foreach loops.

Save as variables_loops.php 
<?php
// Variables and operations
$a = 20;
$b = 6;

echo "a = $a and b = $b<br>";
echo "Addition       : " . ($a + $b) . "<br>";
echo "Subtraction    : " . ($a - $b) . "<br>";
echo "Multiplication : " . ($a * $b) . "<br>";
echo "Division       : " . ($a / $b) . "<br>";
echo "Modulus        : " . ($a % $b) . "<br><br>";

// if - else
if ($a > $b)
  echo "$a is greater than $b<br><br>";
else
  echo "$b is greater than or equal to $a<br><br>";

// while loop
$i = 1;
echo "While loop : ";
while ($i <= 5) {
  echo "$i ";
  $i++;
}
echo "<br>";

// for loop
echo "For loop   : ";
for ($j = 1; $j <= 5; $j++) {
  echo "$j ";
}
echo "<br>";

// foreach loop
$colors = array("Red", "Green", "Blue");
echo "Foreach loop : ";
foreach ($colors as $c) {
  echo "$c ";
}
echo "<br>";
?>

Practical 12 - JavaScript Events

Demonstrate the JavaScript events onclick, onmouseover, onmouseout and onselect.

Save as events.html 
<!DOCTYPE html>
<html>
<head>
<title>JavaScript Events</title>
<script>
function clickEvent() {
  document.getElementById("demo").innerHTML = "Button Clicked!";
}

function mouseIn() {
  document.getElementById("demo").innerHTML = "Mouse Entered the box";
  document.getElementById("box").style.backgroundColor = "lightgreen";
}

function mouseOut() {
  document.getElementById("demo").innerHTML = "Mouse Left the box";
  document.getElementById("box").style.backgroundColor = "lightblue";
}

function selectEvent() {
  var text = document.getElementById("myText");
  document.getElementById("demo").innerHTML = "Selected Text: " +
    text.value.substring(text.selectionStart, text.selectionEnd);
}
</script>
</head>
<body>

<h2>JavaScript Events</h2>

<button onclick="clickEvent()">Click Me</button>

<div id="box" style="width: 250px; height: 80px; background: lightblue; border: 2px solid navy; margin-top: 20px; padding: 10px;"
     onmouseover="mouseIn()"
     onmouseout="mouseOut()">
Mouse Over Me
</div>

<br><br>
<input type="text" id="myText" value="Select this text" onselect="selectEvent()">

<p id="demo"></p>

</body>
</html>
